// DEMO OTP VARIABLES let generatedOTP = ""; let otpExpiry = 0; let otpVerified = false; let otpAttempts = 0; // SHOW FORGOT PASSWORD function showForgotPassword() { document.getElementById("loginSection").style.display = "none"; document.getElementById("resetSection").style.display = "block"; document.getElementById("resetForm").reset(); document.getElementById("resetUsername").value = document.getElementById("username").value.trim(); document.getElementById("usernameStep").style.display = "block"; document.getElementById("otpStep").style.display = "none"; document.getElementById("passwordStep").style.display = "none"; document.getElementById("resetMessage").textContent = ""; generatedOTP = ""; otpExpiry = 0; otpVerified = false; otpAttempts = 0; } // BACK TO LOGIN function showLogin() { document.getElementById("resetSection").style.display = "none"; document.getElementById("loginSection").style.display = "block"; document.getElementById("message").textContent = ""; document.getElementById("resetMessage").textContent = ""; document.getElementById("resetForm").reset(); generatedOTP = ""; otpVerified = false; otpAttempts = 0; } // SEND DEMO OTP function sendDemoOTP() { const username = document.getElementById("resetUsername").value.trim(); const message = document.getElementById("resetMessage"); if (!username) { message.style.color = "red"; message.textContent = "Please enter your registered username."; return; } if (username.toLowerCase() === "admin") { message.style.color = "red"; message.textContent = "Default admin password cannot be reset using demo OTP."; return; } let users; try { users = JSON.parse(localStorage.getItem("assetUsers") || "[]"); if (!Array.isArray(users)) { users = []; } } catch (error) { message.style.color = "red"; message.textContent = "Unable to read registered users."; return; } const user = users.find(item => item.username && item.username.toLowerCase() === username.toLowerCase() ); if (!user) { message.style.color = "red"; message.textContent = "Registered username not found."; return; } // Generate a six-digit DEMO OTP generatedOTP = String( Math.floor(100000 + Math.random() * 900000) ); // OTP expires after 5 minutes otpExpiry = Date.now() + 5 * 60 * 1000; otpVerified = false; otpAttempts = 0; document.getElementById("otpInput").value = ""; document.getElementById("newPassword").value = ""; document.getElementById("confirmPassword").value = ""; document.getElementById("otpStep").style.display = "block"; document.getElementById("passwordStep").style.display = "none"; message.style.color = "#087f8c"; message.textContent = "DEMO OTP: " + generatedOTP + " | Valid for 5 minutes. This OTP is displayed for project demonstration only."; } // VERIFY DEMO OTP function verifyDemoOTP() { const enteredOTP = document.getElementById("otpInput").value.trim(); const message = document.getElementById("resetMessage"); if (!generatedOTP) { message.style.color = "red"; message.textContent = "Please generate an OTP first."; return; } if (Date.now() > otpExpiry) { generatedOTP = ""; otpVerified = false; message.style.color = "red"; message.textContent = "OTP expired. Please request a new OTP."; return; } if (otpAttempts >= 5) { generatedOTP = ""; otpVerified = false; message.style.color = "red"; message.textContent = "Too many incorrect attempts. Request a new OTP."; return; } if (!/^\d{6}$/.test(enteredOTP) || enteredOTP !== generatedOTP) { otpAttempts++; message.style.color = "red"; message.textContent = "Invalid OTP. Attempts remaining: " + (5 - otpAttempts); if (otpAttempts >= 5) { generatedOTP = ""; } return; } otpVerified = true; generatedOTP = ""; document.getElementById("passwordStep").style.display = "block"; message.style.color = "green"; message.textContent = "OTP verified successfully. You can now change your password."; } // SAVE NEW PASSWORD ONLY AFTER OTP VERIFICATION document.getElementById("resetForm").addEventListener( "submit", function(event) { event.preventDefault(); const message = document.getElementById("resetMessage"); if (!otpVerified) { message.style.color = "red"; message.textContent = "Please verify the OTP before changing your password."; return; } const username = document.getElementById("resetUsername").value.trim(); const newPassword = document.getElementById("newPassword").value; const confirmPassword = document.getElementById("confirmPassword").value; if (Date.now() > otpExpiry) { otpVerified = false; message.style.color = "red"; message.textContent = "OTP expired. Please restart password recovery."; return; } if (newPassword.length < 6) { message.style.color = "red"; message.textContent = "Password must contain at least 6 characters."; return; } if (newPassword !== confirmPassword) { message.style.color = "red"; message.textContent = "Passwords do not match."; return; } let users; try { users = JSON.parse(localStorage.getItem("assetUsers") || "[]"); if (!Array.isArray(users)) { users = []; } } catch (error) { message.style.color = "red"; message.textContent = "Unable to read registered users."; return; } const userIndex = users.findIndex(item => item.username && item.username.toLowerCase() === username.toLowerCase() ); if (userIndex === -1) { otpVerified = false; message.style.color = "red"; message.textContent = "Registered user not found."; return; } if (users[userIndex].password === newPassword) { message.style.color = "red"; message.textContent = "Please choose a different password."; return; } users[userIndex].password = newPassword; try { localStorage.setItem("assetUsers", JSON.stringify(users)); } catch (error) { message.style.color = "red"; message.textContent = "Unable to save the password."; return; } otpVerified = false; generatedOTP = ""; otpExpiry = 0; message.style.color = "green"; message.textContent = "Password updated successfully! Please log in with your new password."; document.getElementById("username").value = username; document.getElementById("password").value = ""; document.getElementById("resetForm").reset(); document.getElementById("usernameStep").style.display = "block"; document.getElementById("otpStep").style.display = "none"; document.getElementById("passwordStep").style.display = "none"; } );
